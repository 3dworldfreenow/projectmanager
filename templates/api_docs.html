ESP32 — send readings over HTTP (simplest, works through any network)

Arduino IDE / PlatformIO. Needs ArduinoJson (Library Manager). Posts every 5 minutes, deep-sleeps in between.

#include <WiFi.h>
#include <HTTPClient.h>
#include <WiFiClientSecure.h>
#include <ArduinoJson.h>

const char* WIFI_SSID = "your-wifi";
const char* WIFI_PASS = "your-password";
const char* API_URL   = "{{base_url}}/api/v1/data";
const char* DEVICE_KEY = "{{key}}";
const int   SLEEP_MIN  = 5;

float readTemp()     { return 24.6; }   // replace with BME280 etc.
float readHumidity() { return 61.2; }
float readPressure() { return 1008.3; }

void setup() {
  Serial.begin(115200);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  for (int i = 0; i < 40 && WiFi.status() != WL_CONNECTED; i++) delay(500);
  if (WiFi.status() == WL_CONNECTED) {
    WiFiClientSecure client; client.setInsecure();   // or client.setCACert(rootCA) for strict TLS
    HTTPClient http; http.begin(client, API_URL);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("X-Device-Key", DEVICE_KEY);

    StaticJsonDocument<256> doc;
    doc["temp"] = readTemp(); doc["hum"] = readHumidity(); doc["pressure"] = readPressure();
    doc["battery_v"] = analogReadMilliVolts(34) * 2 / 1000.0;
    String body; serializeJson(doc, body);

    int code = http.POST(body);
    Serial.printf("POST -> %d %s\n", code, http.getString().c_str());
    // the response contains any queued commands: {"ok":true,"commands":[{"cmd":"relay","value":1}]}
    http.end();
  }
  esp_sleep_enable_timer_wakeup((uint64_t)SLEEP_MIN * 60ULL * 1000000ULL);
  esp_deep_sleep_start();
}
void loop() {}

ESP32 — MQTT (live, two-way, best for mains-powered nodes)

Needs PubSubClient + ArduinoJson. Publishes to rainflow/<key>/data, listens on rainflow/<key>/cmd. Broker: {{mqtt_host}}:{{mqtt_port}} {% if not mqtt.host %}broker not configured yet — see "Set up the MQTT broker"{% elif mqtt.connected %}server connected{% else %}server not connected{% if mqtt.error %} · {{mqtt.error}}{% endif %}{% endif %}

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

const char* WIFI_SSID = "your-wifi";  const char* WIFI_PASS = "your-password";
const char* MQTT_HOST = "{{mqtt_host}}"; const int MQTT_PORT = {{mqtt_port}};
const char* MQTT_USER = "rainflow";   const char* MQTT_PASS = "mqtt-password";   // as set on the broker
const char* DEVICE_KEY = "{{key}}";
String topicData = String("rainflow/") + DEVICE_KEY + "/data";
String topicCmd  = String("rainflow/") + DEVICE_KEY + "/cmd";

WiFiClient net; PubSubClient mqtt(net);
unsigned long lastSend = 0;

void onCommand(char* topic, byte* payload, unsigned int len) {
  StaticJsonDocument<256> doc; deserializeJson(doc, payload, len);
  const char* cmd = doc["cmd"] | "";
  if (strcmp(cmd, "relay") == 0) digitalWrite(26, doc["value"] | 0);   // example: {"cmd":"relay","value":1}
  if (strcmp(cmd, "interval") == 0) { /* change send interval */ }
}
void connectMqtt() {
  while (!mqtt.connected()) {
    if (mqtt.connect(DEVICE_KEY, MQTT_USER, MQTT_PASS)) mqtt.subscribe(topicCmd.c_str());
    else delay(2000);
  }
}
void setup() {
  Serial.begin(115200); pinMode(26, OUTPUT);
  WiFi.begin(WIFI_SSID, WIFI_PASS); while (WiFi.status() != WL_CONNECTED) delay(300);
  mqtt.setServer(MQTT_HOST, MQTT_PORT); mqtt.setCallback(onCommand);
}
void loop() {
  if (!mqtt.connected()) connectMqtt();
  mqtt.loop();
  if (millis() - lastSend > 10000) {               // every 10 s
    lastSend = millis();
    StaticJsonDocument<256> doc;
    doc["temp"] = 24.6; doc["hum"] = 61.2; doc["wind_ms"] = 3.4; doc["rain_mm"] = 0.2;
    char buf[256]; serializeJson(doc, buf);
    mqtt.publish(topicData.c_str(), buf);
  }
}

Receiving commands on the device

From the Live data page (or POST /api/v1/command) you send e.g. {"cmd":"relay","value":1}. It reaches the device two ways:

  • MQTT — instantly on rainflow/<key>/cmd (see onCommand above).
  • HTTP — included in the response of every POST /api/v1/data (commands:[…]), or poll GET /api/v1/commands. Good for battery nodes that wake every few minutes.

HTTP API reference

POST/api/v1/datadevice → server
Authheader X-Device-Key: {{key}} (or ?key=)
BodyJSON object of fields, e.g. {"temp":22.5,"hum":61}. Optional "ts":"2026-10-31 14:00:00" for buffered readings. Form-encoded also accepted.
Returns{"ok":true,"stored":["temp","hum"],"commands":[…]}
curl -X POST {{base_url}}/api/v1/data \
  -H "X-Device-Key: {{key}}" -H "Content-Type: application/json" \
  -d '{"temp":22.5,"hum":61,"pressure":1008.3}'
GET/api/v1/commandsdevice polls for queued commands (marks them delivered)
POST/api/v1/commandserver → device · logged-in users · body {"device":1,"cmd":"relay","value":1}

Python (Raspberry Pi gateway, scripts)

import requests, time
URL = "{{base_url}}/api/v1/data"; KEY = "{{key}}"
while True:
    r = requests.post(URL, json={"temp": 22.5, "hum": 61}, headers={"X-Device-Key": KEY}, timeout=10)
    for c in r.json().get("commands", []): print("command:", c)
    time.sleep(60)

Set up the MQTT broker (once, on the server)

# Mosquitto broker on the VPS
yum install -y mosquitto || apt install -y mosquitto
cat > /etc/mosquitto/conf.d/rainflow.conf <<'EOF'
listener 1883
allow_anonymous false
password_file /etc/mosquitto/passwd
EOF
mosquitto_passwd -c -b /etc/mosquitto/passwd rainflow 'mqtt-password'
systemctl enable --now mosquitto
firewall-cmd --permanent --add-port=1883/tcp 2>/dev/null && firewall-cmd --reload   # (and in the Contabo panel firewall)

# tell Rainflow about it
uv pip install --python /opt/rainflow/projectmanager/.venv paho-mqtt
sed -i 's|^Environment=.*|& MQTT_HOST=127.0.0.1 MQTT_PORT=1883 MQTT_USER=rainflow MQTT_PASS=mqtt-password|' /etc/systemd/system/rainflow.service
systemctl daemon-reload && systemctl restart rainflow

# test from any machine
mosquitto_pub -h {{mqtt_host}} -u rainflow -P 'mqtt-password' -t rainflow/{{key}}/data -m '{"temp":21.5}'

Devices connect to {{mqtt_host}}:1883 with the user/password above. For TLS (port 8883) add certfile/keyfile lines pointing at the Let's Encrypt certificate and set MQTT_TLS=1.
